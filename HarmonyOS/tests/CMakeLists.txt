cmake_minimum_required(VERSION 3.16)
project(btmobile_host_tests LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../native-deps")
if(EXISTS "${DEPS}/boost/boost/version.hpp")
  set(Boost_NO_SYSTEM_PATHS ON)
  set(BOOST_ROOT "${DEPS}/boost")
endif()
set(deprecated-functions OFF CACHE BOOL "" FORCE)
set(build_tests OFF CACHE BOOL "" FORCE)
set(build_examples OFF CACHE BOOL "" FORCE)
set(build_tools OFF CACHE BOOL "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(webtorrent OFF CACHE BOOL "" FORCE)
find_package(OpenSSL REQUIRED)
add_compile_definitions(TORRENT_HAVE_MMAP=0)
add_subdirectory("${DEPS}/libtorrent" "${CMAKE_BINARY_DIR}/libtorrent")
find_package(LibArchive REQUIRED)
add_executable(settings-test settings_test.cpp)
target_include_directories(settings-test PRIVATE "${DEPS}/headers")
target_compile_options(settings-test PRIVATE -UNDEBUG)
add_executable(core-test core_test.cpp)
target_compile_options(core-test PRIVATE -UNDEBUG)
target_include_directories(core-test PRIVATE "${DEPS}/headers")
target_link_libraries(core-test PRIVATE torrent-rasterbar LibArchive::LibArchive OpenSSL::SSL OpenSSL::Crypto)
if(DEFINED ENV{OHOS_NDK_HOME})
  configure_file("$ENV{OHOS_NDK_HOME}/sysroot/usr/include/network/netmanager/net_connection_type.h"
    "${CMAKE_CURRENT_BINARY_DIR}/sdk-types/network/netmanager/net_connection_type.h" COPYONLY)
  add_executable(network-test network_test.cpp)
  target_compile_options(network-test PRIVATE -UNDEBUG)
  target_include_directories(network-test PRIVATE "${CMAKE_CURRENT_BINARY_DIR}/sdk-types")
  target_link_libraries(network-test PRIVATE torrent-rasterbar)
endif()
