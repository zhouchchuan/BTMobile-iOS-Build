cmake_minimum_required(VERSION 3.16)
project(btmobile LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
set(DEPS "${CMAKE_CURRENT_SOURCE_DIR}/../../../../native-deps")
list(PREPEND CMAKE_PREFIX_PATH "${DEPS}/prefix")
set(Boost_NO_SYSTEM_PATHS ON)
set(BOOST_ROOT "${DEPS}/boost")
set(Boost_INCLUDE_DIR "${DEPS}/boost" CACHE PATH "Cross-compiled Boost headers" FORCE)
set(OPENSSL_ROOT_DIR "${DEPS}/prefix")
set(OPENSSL_USE_STATIC_LIBS TRUE)
set(OPENSSL_INCLUDE_DIR "${DEPS}/prefix/include" CACHE PATH "" FORCE)
set(OPENSSL_SSL_LIBRARY "${DEPS}/prefix/lib/libssl.a" CACHE FILEPATH "" FORCE)
set(OPENSSL_CRYPTO_LIBRARY "${DEPS}/prefix/lib/libcrypto.a" CACHE FILEPATH "" FORCE)
find_package(OpenSSL REQUIRED)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(deprecated-functions OFF CACHE BOOL "" FORCE)
set(build_tests OFF CACHE BOOL "" FORCE)
set(build_examples OFF CACHE BOOL "" FORCE)
set(build_tools OFF CACHE BOOL "" FORCE)
set(python-bindings OFF CACHE BOOL "" FORCE)
set(webtorrent OFF CACHE BOOL "" FORCE)
add_compile_definitions(TORRENT_HAVE_MMAP=0 BTMOBILE_OHOS=1)
add_subdirectory("${DEPS}/libtorrent" "${CMAKE_BINARY_DIR}/libtorrent")
target_include_directories(torrent-rasterbar PRIVATE "${CMAKE_CURRENT_SOURCE_DIR}")
target_link_libraries(torrent-rasterbar PUBLIC libnet_connection.so)
add_library(btmobile SHARED bridge.cpp)
include("${DEPS}/../ci/bit7z.cmake")
add_library(sevenzip SHARED IMPORTED)
set_target_properties(sevenzip PROPERTIES IMPORTED_LOCATION "${DEPS}/prefix/lib/lib7zip.so")
target_link_libraries(btmobile PRIVATE bit7z -Wl,--no-as-needed sevenzip -Wl,--as-needed)
target_include_directories(btmobile PRIVATE "${DEPS}/headers" "${DEPS}/prefix/include")
target_link_libraries(btmobile PRIVATE torrent-rasterbar "${DEPS}/prefix/lib/libarchive.a" "${DEPS}/prefix/lib/liblzma.a" "${DEPS}/prefix/lib/libz.a" OpenSSL::SSL OpenSSL::Crypto libace_napi.z.so libhilog_ndk.z.so)
target_compile_options(btmobile PRIVATE -Wall -Wextra)

# Keep the AV1 compatibility decoder separate from the established BT core.
set(MEDIA_PREFIX "${DEPS}/media/prefix")
add_library(btmedia SHARED media_bridge.cpp)
target_include_directories(btmedia PRIVATE "${DEPS}/headers" "${MEDIA_PREFIX}/include")
foreach(component avformat avcodec avutil swresample swscale)
    add_library(media_${component} SHARED IMPORTED)
    set_target_properties(media_${component} PROPERTIES IMPORTED_LOCATION "${MEDIA_PREFIX}/lib/lib${component}.so")
    target_link_libraries(btmedia PRIVATE media_${component})
endforeach()
target_link_libraries(btmedia PRIVATE libace_napi.z.so libhilog_ndk.z.so libnative_window.so libEGL.so libGLESv3.so libohaudio.so)
target_compile_options(btmedia PRIVATE -Wall -Wextra)
